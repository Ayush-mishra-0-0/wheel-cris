SELECT TOP 20
    wsm.wsmId,
    wsm.wsmUpdatedOn AS MeasurementDate,
    emr.EmrSerialNo AS WheelsetSerialNo,
    lm.LomNumber AS LocoNumber,
    lm.LomType AS LocoType,
    lm.LomIntialWheelDiaBogie1 AS InitialDiaBogie1,
    wsm.wsmDia1 AS LeftDiameter_mm,
    wsm.wsmDia2 AS RightDiameter_mm,
    wsm.wsmFlangeThickness1 AS FlangeThick_mm,
    wsm.wsmturning1 AS TurningFlag,
    rtis.RlkdTotalDistance AS RecentTotalDistance_km,
    rtis.RlkdReportDate AS MileageReportDate
FROM WheelSetMeasurements wsm
INNER JOIN EquipmentMasterRegister emr 
    ON wsm.wsmEquipmentId = emr.EmrId
LEFT JOIN LocoEquipments le 
    ON emr.EmrId = le.LoeEquipmentMasterRegister
LEFT JOIN LocoMaster lm 
    ON le.LoeLocoMaster = lm.LomId
OUTER APPLY (
    SELECT TOP 1 RlkdTotalDistance, RlkdReportDate
    FROM RtisLocoKmDetails r
    WHERE r.RlkdLocoNumber = lm.LomNumber
      AND r.RlkdReportDate <= wsm.wsmUpdatedOn
    ORDER BY r.RlkdReportDate DESC
) rtis
ORDER BY wsm.wsmUpdatedOn DESC;
